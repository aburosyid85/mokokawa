Konektor Backend Google Apps Script (Code.gs)

Tempelkan fungsi-fungsi di bawah ini ke bagian paling bawah file Code.gs Anda (tanpa menghapus kode yang sudah ada). Script ini sudah mendukung REST API JSON (doGet & doPost) untuk dihubungkan ke Vercel sekaligus tetap mendukung Web App GAS: 

// ====================================================================================
// FUNGSI HYBRID API (VERCEL REST API + GOOGLE APPS SCRIPT WEB APP)
// ====================================================================================
function doGet(e) {
  // Jika dipanggil dari Vercel menggunakan parameter ?action=getData, kembalikan JSON (CORS-ready)
  if (e && e.parameter && e.parameter.action === "getData") {
    var payload = getMokokawaData();
    return ContentService.createTextOutput(JSON.stringify(payload))
      .setMimeType(ContentService.MimeType.JSON);
  }

  // Fallback jika dibuka langsung dari link Web App Google Apps Script
  return HtmlService.createHtmlOutputFromFile('Index')
    .setTitle('MOKOKAWA 2.1 - KPPN Luwuk')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL)
    .addMetaTag('viewport', 'width=device-width, initial-scale=1');
}

// Endpoint POST untuk menyimpan konfigurasi Relaksasi IKPA & Upload File Excel langsung dari Aplikasi Web
function doPost(e) {
  try {
    var body = JSON.parse(e.postData.contents);
    if (body && body.action === "saveRelaksasi") {
      var res = saveRelaksasiConfigToSheet(body.config);
      return ContentService.createTextOutput(JSON.stringify(res))
        .setMimeType(ContentService.MimeType.JSON);
    }
    if (body && (body.action === "uploadExcel" || (body.files && Array.isArray(body.files)))) {
      var resUpload = uploadAndProcessExcelFiles(body.files);
      return ContentService.createTextOutput(JSON.stringify(resUpload))
        .setMimeType(ContentService.MimeType.JSON);
    }
    return ContentService.createTextOutput(JSON.stringify({ success: false, error: "Invalid action" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ success: false, error: err.message }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

// Fungsi penerima upload file Excel (Mendukung Upload Manual Web App & Auto-Sync Tampermonkey)
function uploadAndProcessExcelFiles(filesArray) {
  var excelFolderId = "1DkY1aMqEBxK_pRCYgjysy96wpa-cli5t";
  var folder = DriveApp.getFolderById(excelFolderId);
  var savedNames = [];

  if (!filesArray || !filesArray.length) {
    return { success: false, error: "Tidak ada file yang dikirim." };
  }

  for (var i = 0; i < filesArray.length; i++) {
    var f = filesArray[i];
    // Dukung properti base64Data (dari Web App) maupun base64 (dari Tampermonkey)
    var rawBase64 = f.base64Data || f.base64 || "";
    if (!rawBase64) continue;

    // Normalisasi nama file berdasarkan kode 'jenis' dari Tampermonkey agar 100% cocok dengan Aturan 1 & Aturan 2 di updateMasterSheetFromExcel()
    var finalFileName = f.fileName || "Upload_Mokokawa.xlsx";
    var jenisKode = String(f.jenis || "").toUpperCase();

    if (jenisKode === "HAL3_DASHBOARD" && finalFileName.toUpperCase().indexOf("RPD") === -1) {
      finalFileName = "RPD_" + finalFileName;
    } else if (jenisKode === "HAL3_DETAIL" && finalFileName.toUpperCase().indexOf("DETAIL INDIKATOR HALAMAN 3 DIPA") === -1) {
      finalFileName = "DETAIL INDIKATOR HALAMAN 3 DIPA_" + finalFileName;
    }

    var bytes = Utilities.base64Decode(rawBase64);
    var blob = Utilities.newBlob(bytes, "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet", finalFileName);
    folder.createFile(blob);
    savedNames.push(finalFileName);
  }

  // Langsung jalankan pemrosesan ke Master Database & hitung ulang IKPA Deviasi
  updateMasterSheetFromExcel();

  return {
    success: true,
    processedFiles: savedNames,
    updatedData: getMokokawaData()
  };
}

function getMokokawaData() {
  var masterSheetId = "19zzOnuz0gU0fVwmMMIM_P0RDFi3wxekY1Dc9OuOG-KU";
  var ss = SpreadsheetApp.openById(masterSheetId);

  var sheet1 = ss.getSheets()[0];
  var ikpaDevSheet = ss.getSheetByName("IKPA Deviasi");
  var ikpaKppnSheet = ss.getSheetByName("IKPA KPPN");
  var logSheet = ss.getSheetByName("Log_Update");
  var configSheet = ss.getSheetByName("Config_Relaksasi");

  var logTime = logSheet ? logSheet.getRange(2, 1).getDisplayValue() : "-";

  var savedRelaksasiConfig = null;
  if (configSheet) {
    var rawJson = configSheet.getRange(2, 1).getValue();
    if (rawJson) {
      try {
        savedRelaksasiConfig = JSON.parse(rawJson);
      } catch (err) {
        savedRelaksasiConfig = null;
      }
    }
  }

  function sheetToObjects(sheet) {
    if (!sheet) return [];
    var vals = sheet.getDataRange().getDisplayValues();
    if (vals.length < 2) return [];
    var headers = vals[0];
    var result = [];
    for (var r = 1; r < vals.length; r++) {
      var obj = {};
      for (var c = 0; c < headers.length; c++) {
        obj[headers[c]] = vals[r][c];
      }
      result.push(obj);
    }
    return result;
  }

  return {
    logUpdate: logTime,
    currentMonth: new Date().getMonth() + 1,
    relaksasiConfig: savedRelaksasiConfig,
    sheet1: sheetToObjects(sheet1),
    ikpaDeviasi: sheetToObjects(ikpaDevSheet),
    ikpaKppn: sheetToObjects(ikpaKppnSheet)
  };
}

function saveRelaksasiConfigToSheet(configObj) {
  var masterSheetId = "19zzOnuz0gU0fVwmMMIM_P0RDFi3wxekY1Dc9OuOG-KU";
  var ss = SpreadsheetApp.openById(masterSheetId);
  var configSheet = ss.getSheetByName("Config_Relaksasi");
  if (!configSheet) {
    configSheet = ss.insertSheet("Config_Relaksasi");
  }
  configSheet.clear();
  configSheet.getRange(1, 1).setValue("JSON_Config_Relaksasi");
  configSheet.getRange(1, 2).setValue("Terakhir_Diubah");
  configSheet.getRange(2, 1).setValue(JSON.stringify(configObj));
  configSheet.getRange(2, 2).setValue(Utilities.formatDate(new Date(), "Asia/Makassar", "yyyy-MM-dd HH:mm:ss"));
  return { success: true };
}

// FUNGSI INSTALASI TRIGGER INSTAN SAAT UPLOAD FILE DI GOOGLE FORM (JALANKAN 1 KALI)
// ====================================================================================
function pasangTriggerOtomatisForm() {
  var formIdTarget = "1g7GOig7UL29TV-NT3hidYuLdmFWgXDh3wl4r-bEBJB8";
  var triggers = ScriptApp.getProjectTriggers();
  for (var i = 0; i < triggers.length; i++) {
    ScriptApp.deleteTrigger(triggers[i]);
  }
  var form = FormApp.openById(formIdTarget);
  ScriptApp.newTrigger("prosesSaatFormDiSubmit")
    .forForm(form)
    .onFormSubmit()
    .create();
  Logger.log("Trigger otomatis saat upload Google Form berhasil dipasang!");
}

function prosesSaatFormDiSubmit(e) {
  var folderTujuanId = "1DkY1aMqEBxK_pRCYgjysy96wpa-cli5t";
  var folderTujuan = DriveApp.getFolderById(folderTujuanId);
  var adaFile = false;

  if (e && e.response) {
    var itemResponses = e.response.getItemResponses();
    for (var i = 0; i < itemResponses.length; i++) {
      var jawaban = itemResponses[i].getResponse();
      var fileIds = Array.isArray(jawaban) ? jawaban : [jawaban];
      for (var j = 0; j < fileIds.length; j++) {
        var fId = String(fileIds[j]).trim();
        if (fId.length >= 20) {
          try {
            DriveApp.getFileById(fId).moveTo(folderTujuan);
            adaFile = true;
          } catch (err) {
            Logger.log("Gagal memindahkan file: " + err.message);
          }
        }
      }
    }
  }
  updateMasterSheetFromExcel();
}
